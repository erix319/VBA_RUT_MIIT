    Dim salesAugust() As Double

    Dim i As Integer

Sub AverageYieldWithoutArray()

    Dim yields() As Double
    Dim totalYield As Double
    Dim totalArea As Double
    Dim i As Integer
    
    ' Пример данных
    areas = Array(10.5, 20.0, 15.5) ' Замените на реальные данные
